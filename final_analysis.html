After basic import, we will immediately export the network of gene-trait connections for modularity analysis. This is performed conditionally, as the process is relatively long.

human_map = data.frame(gene = NA, term = NA)
mouse_map = data.frame(gene = NA, term = NA)
export_network = FALSE
if (export_network) {
  for (i in 1:nrow(merged_data)) {
    print(i)
    human_gene = merged_data[i, "gene_human"]
    mouse_gene = merged_data[i, "gene_mouse"]
    murine_terms = merged_data[i, "MP_high_level"]
    human_terms = merged_data[i, "HP_high_level"]
    if (murine_terms != "") {
      term_set = str_match_all(murine_terms, "MP:\\d+")[[1]][, 1]
      for (tg_term in term_set) {
        mouse_map <- rbind(mouse_map, c(mouse_gene, tg_term))
      }
    }
    if (human_terms != "") {
      term_set = str_match_all(human_terms, "MP:\\d+")[[1]][, 1]
      for (tg_term in term_set) {
        human_map <- rbind(human_map, c(human_gene, tg_term))
      }
    }
  }
  write.table(na.omit(human_map), file='hpo_netowrk.txt', sep='\t', 
              col.names=F, row.names=F, quote=F)
  write.table(na.omit(mouse_map), file='mgd_netowrk.txt', sep='\t', 
              col.names=F, row.names=F, quote=F)
}

  • Does the conversion of HPO terms to MP introduce substantial bias?

  • Do upper-level terms perform well in reducing spurious pleiotropy arising from correlated traits?

Finally, we make a master function to plot all features we may want to, with the following imaportant args:

  • axis_lab, ax_lim, log_scale - regulating the axis properties on plots [beware - axis limit is applied before plotting];

  • mouse_feature - specification of a separate column for the value of the feature in mice (if this is different from the main feature column);

  • gwas_data_file - filename to use for GWAS locus-level data;

  • cutoff, cut_name - parameters for creating binary plots with a cutoff value to be used for splitting feature into categories;

  • use_maxindep, use_locuswide - arguments that, if set, override the set of columns used for analysis, juxtaposing the main (closest gene, clustered data) dataset against the selected one.

plot_feature <- function(feature_name, axis_lab, ax_lim=NA, log_scale=FALSE,
                         mouse_feature=NA, gwas_data_file=NA, 
                         cutoff=NA, cut_name=NA, 
                         use_maxindep=FALSE, use_locuswide=FALSE) {
  if (use_maxindep) {
    source_names <- c('panUKB', 'panUKB_maxindep')
  } else if (use_locuswide) {
    source_names <- c('panUKB', 'panUKB_all')
  }
  pl_col_names <- paste0('pleiotropy_', source_names)
  feat_df <- merged_data[, c(feature_name, pl_col_names)]
  feat_df <- na.omit(melt(feat_df, id.vars=feature_name))
  colnames(feat_df) <- c('feature', 'dataset', 'group')
  feat_df$dataset = sapply(as.character(feat_df$dataset), 
                           function(x) strsplit(x, 'py_')[[1]][2])
  
  if (!is.na(mouse_feature)) {
    # Removing mouse data
    feat_df <- feat_df[feat_df$dataset != 'MGD', ]
    mouse_snippet <- na.omit(data.frame(feature=merged_data[, mouse_feature],
                                dataset='MGD',
                                group=merged_data$pleiotropy_MGD))
    feat_df <- as.data.frame(rbind(feat_df, mouse_snippet))
  }
  
  feat_df$group = factor(as.character(feat_df$group),
                         levels=c('none', 'moderate', 'high'))
  if (!is.na(gwas_data_file)) {
    gwas_values = read.table(gwas_data_file, sep='\t', header=T)
    feat_df <- gwas_values
    colnames(feat_df) <- c('group', 'feature')
    feat_df$group = factor(as.character(feat_df$group), 
                           levels=c('none', 'moderate', 'high'))
    feat_df$dataset <- 'panUKB (locus-wise)'
    source_names <- 'panUKB (locus-wise)'
  }
  
  summary_df <- aggregate(feature~group+dataset, feat_df, median)
  summary_df$pvalue = NA
  for (src_name in source_names) {
    for (tg_group in c('moderate', 'high')) {
      wtres <- wilcox.test(feature~group, feat_df[feat_df$dataset == src_name &
                  feat_df$group %in% c('none', tg_group), ])
      summary_df[summary_df$dataset == src_name  & 
                 summary_df$group == tg_group, 'pvalue'] = wtres$p.value
    }
  }
  summary_df$pvalue = p.adjust(summary_df$pvalue, method='fdr')
  print_sum_df <- summary_df
  
  if (is.na(ax_lim)) {
    ax_lim = max(feat_df$feature)
  } else {
    feat_df <- feat_df[feat_df$feature <= ax_lim, ]
  }
  if (log_scale) {
    line_shift <- ax_lim
  } else {
    line_shift = ax_lim/10
  }
  
  line_data <- data.frame(dataset=rep(source_names, each=4),
                  group=rep(c('none', 'moderate', 'none', 'high'), 
                            length(source_names)),
                  feature = rep(c(ax_lim + line_shift, ax_lim + line_shift, 
                      ax_lim + 2.5 * line_shift, ax_lim + 2.5 * line_shift), 
                      length(source_names)),
                  line_cat = rep(c(1, 1, 2, 2), length(source_names)))
  summary_df$plot_label = ifelse(summary_df$pvalue < 0.05,
                      ifelse(summary_df$pvalue < 0.01,
                      ifelse(summary_df$pvalue < 0.001, '***', '**'), '*'), 
                          'n.s.')
  summary_df$feature = rep(c(NA, ax_lim + 1.75 * line_shift, 
                           ax_lim + 3.25 * line_shift), length(source_names))
  
  # Shifting the asterisks a little
  summary_df$feature = ifelse(summary_df$plot_label == "n.s.",
        summary_df$feature + 0.05 * line_shift, 
        summary_df$feature - 0.25 * line_shift)

  out_plt <- ggplot(feat_df, aes(x=group, y=feature, fill=group)) +
    geom_violin(scale = "width") + theme_bw() + 
    theme(panel.grid = element_blank(), strip.background = my_ggrect) +
    geom_boxplot(outlier.shape=NA, width=0.25, col='black', fill='white') +
    geom_line(data=line_data, aes(group=line_cat)) +
    geom_text(data=summary_df[summary_df$group == 'moderate', ], 
              aes(label=plot_label), nudge_x=-0.5) +
    geom_text(data=summary_df[summary_df$group == 'high', ],
              aes(label=plot_label), nudge_x=-1) +
    ylab(axis_lab) + xlab('Pleiotropy group') +
    facet_wrap(~dataset, nrow=1) +  guides(fill=F) +
    scale_fill_brewer(palette='BuPu') +
    scale_y_continuous(limits=c(0, ax_lim * 1.4))
  
  if (log_scale) {
    out_plt <- out_plt + scale_y_log10()
  }
  
  if (!is.na(cutoff)) {
    cutoff_stats <- aggregate(feature~group+dataset, feat_df, function(x) 
      table(x > cutoff)/length(x))
    chisq_pvals <- c()
    for (src_name in source_names) {
      ds_df <- aggregate(feature~group, feat_df[feat_df$dataset == src_name & 
          feat_df$group %in% c('none', 'high'), ], function(x) table(x > cutoff))
      chisq_pvals <- c(chisq_pvals, chisq.test(ds_df$feature)$p.value)
    }
    cutoff_stats$no <- cutoff_stats$feature[, 1]
    cutoff_stats$yes <- cutoff_stats$feature[, 2]
    cutoff_stats <- cutoff_stats[, c(1, 2, 4, 5)]
    print(cutoff_stats)
    names(chisq_pvals) = source_names
    print(chisq_pvals)
    
    cutoff_df <- melt(cutoff_stats, measure.vars=c('no', 'yes'),
                      id.vars=c('group', 'dataset'))
    prop_plot <- ggplot(cutoff_df, aes(x=group, y=value, fill=variable)) +
      geom_bar(stat='identity', col='black') + theme_bw() +
      xlab('Pleiotropy group') + ylab('Proportion of genes') +
      facet_wrap(~dataset, ncol=4) +
#      coord_flip() + 
#      facet_wrap(~dataset, nrow=4) +
      scale_fill_brewer(palette="OrRd", name=cut_name) +
      theme(strip.background = my_ggrect, 
            panel.grid = element_blank(),
            legend.position = 'top',
            axis.text.x=element_text(angle=45, hjust=1))
    return(prop_plot)
    
  } else {
    print(print_sum_df)
    return(out_plt)
  }
  
}

# Conveniences for EA
make_enrichment_plot <- function(gene_list, orgdb_name,
                                 ontology, plot_title,
                                 data_source="PO",
                                 use_t2g=FALSE,
                                 t2g_df=NA) {
  if (use_t2g) {
    ego <- enricher(gene = gene_list, 
                    TERM2GENE = t2g_df,
                    pvalueCutoff = 0.05,
                    pAdjustMethod = "BH",
                    maxGSSize = 2500)
  } else {
    ego <- enrichGO(gene = gene_list,
                  OrgDb = orgdb_name,
                  keyType = "SYMBOL",
                  ont = ontology,
                  pvalueCutoff = 0.05,
                  pAdjustMethod = "BH",
                  maxGSSize = 2500,
                  readable = TRUE)
  }
  
  go_res_plot <- dotplot(ego, showCategory=10, title=plot_title)
  write.csv(ego, file = paste0(data_source, '_', orgdb_name, "_", ontology, ".csv"), 
            row.names = FALSE)
  return(list('df' = as.data.frame(ego), 'plot' = go_res_plot))
}

Independent large pathways

Now let’s try constructing the maximum independent set of CP pathways. To do so, we will do the following:

  1. Compute the number of genes per pathway;
  2. Take the largest one by the total gene count;
  3. Subtract genes belonging to the pathway in (2), return to step (1).
  4. Repeat until a certain number or fraction of genes are covered
max_indep_gs <- function(t2g_df, gene_cutoff=15000,
                         increment_cutoff=50) {
  next_iteration = TRUE
  indep_gs_set <- c()
  seen_genes <- c()
  while (next_iteration) {
    gs_gene_counts = table(t2g_df$gs_name)
    largest_gs <- names(gs_gene_counts)[which.max(gs_gene_counts)]
    largest_gs_genes <- t2g_df[t2g_df$gs_name == largest_gs, 
                               'gene_symbol']
    seen_genes <- c(seen_genes, largest_gs_genes)
    indep_gs_set <- c(indep_gs_set, largest_gs)
    print(paste(largest_gs, length(largest_gs_genes), 
          length(indep_gs_set), length(seen_genes), sep=' '))
    t2g_df <- t2g_df[!(t2g_df$gene_symbol %in% largest_gs_genes), ]
    if (length(seen_genes) >= gene_cutoff || 
        nrow(t2g_df) == 0 ||
        length(largest_gs_genes) < increment_cutoff) {
      break
    }
  }
  return(indep_gs_set)
}

Analysis of per-term pleiotropic genes fraction

This analysis aims at looking into how well do the percentages of highly pleiotropic genes according to HPO and pan-UKB correlate within a given GO term. To this end, we define a set of convenience functions (note that the function has the remove_common_genes parameter that aims to exclude genes that are highly pleiotropic in both sources, analogous to manipulations for Venn diagrams above).

get_share_perc <- function(go_term_id, gotab_df,
                           remove_common_genes) {
  snippet <- gotab_df[gotab_df$GO == go_term_id, ]
  total_uses <- nrow(snippet)
  if (remove_common_genes) {
    snippet <- snippet[!(snippet$pleiotropy_HPO == 'high' &
                         snippet$pleiotropy_panUKB == 'high'), ]
  }
  all_hpo_uses <- sum(snippet$HP_high_count > 0, na.rm=T)
  all_puk_uses <- sum(snippet$pleio_degree > 0, na.rm=T)
  all_any_uses <- sum(snippet$HP_high_count > 0 |
                      snippet$pleio_degree > 0, na.rm=T)
  high_hpo_uses <- sum(snippet$pleiotropy_HPO == 'high', na.rm = T)
  high_puk_uses <- sum(snippet$pleiotropy_panUKB == 'high', na.rm = T)
  return(c(total_uses, c(all_hpo_uses, all_puk_uses, all_any_uses,
           high_hpo_uses, high_puk_uses)/total_uses))
}

annotate_gene_percent <- function(enrich_df) {
  enrich_df$k <- sapply(enrich_df$GeneRatio, function(x)
    as.numeric(strsplit(x, '/')[[1]][1]))
  enrich_df$K <- sapply(enrich_df$BgRatio, function(x)

  enrich_df$perc_enrich = enrich_df$k/enrich_df$K
  return(enrich_df)
}

Analysis of the natural selection signatures

Finally, we turn to the analysis of natural selection effects for genes in different groups of pleiotropy. There are several major sources of selection statisitcs used in the analysis:

  • phastCons scores (downloaded from the UCSC Table Browser);

  • constraint metrics derived from gnomAD (LOEUF and pLI, taken according to gnomAD v.2.1);

  • iHS scores for the British population (taken from the study by Johson et al., 2018, Nat Ecol Evol) [the choice of GBR as the target population was motivated by similarity to the UKB data];

  • DRC150 (density of recent coalescent events) according to the study of Palamara et al., 2019, Nat Genet.
